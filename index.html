Google Sheets Sync
▾

Connect to a Google Sheet so your data syncs across all your devices. Follow the setup guide below to get your Web App URL, then paste it here.

Google Sheet Sync Controls
▾

Push sends all local data to the sheet — always do this after making changes.
Pull replaces all local data with what is in the sheet — use when switching devices.
Clear local data wipes this device only — your sheet data is unaffected.

Google Sheets Setup Guide
▾

Follow these steps once to connect your Google Sheet:

  1. Go to sheets.google.com and create a new blank spreadsheet. Name it Ledger.
  2. Click Extensions → Apps Script in the menu bar.
  3. Delete any existing code in the editor, then paste in the Apps Script code below.
  4. Click Save (the floppy disk icon), then click Deploy → New deployment.
  5. Set type to Web app. Set "Execute as" to Me and "Who has access" to Anyone. Click Deploy.
  6. Copy the Web app URL it gives you and paste it into the field above.
  7. Click Connect, then Push to Google Sheets to do your first sync.
Apps Script Code — paste this into the editor
const SHEET_NAME = 'Transactions';
const LEARNED_SHEET = 'Learned';
const INCOME_SHEET = 'Income';
const SAVINGS_SHEET = 'Savings';
const CONTRIB_SHEET = 'Contributions';
const UPLOADS_SHEET = 'Uploads';
const RENTAL_SHEET = 'Rental';
const CATS_SHEET = 'Categories';
const SETTINGS_SHEET = 'Settings';

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (data.action === 'push') {
      writeSheet(ss, SHEET_NAME, ['id','date','merchant','amount','category','person','card','note','refund','manual','incomeType','savingsAccountId','incomeCategory','needs','rental','label','recurring'],
        (data.txns||[]).map(t => [t.id,t.date,t.merchant,t.amount,t.category,t.person||'',t.card||'',t.note||'',t.refund?'true':'false',t.manual?'true':'false',t.incomeType||'',t.savingsAccountId||'',t.incomeCategory||'',t.needs===true?'true':t.needs===false?'false':'',t.rental?'true':'',t.label||'',t.recurring?'true':'']));
      writeSheet(ss, LEARNED_SHEET, ['merchant','category'],
        Object.entries(data.learned||{}).map(([m,c]) => [m,c]));
      writeSheet(ss, INCOME_SHEET, ['id','date','amount','type','person','desc','note','fromTxnId','category'],
        (data.incomes||[]).map(i => [i.id,i.date,i.amount,i.type,i.person||'',i.desc||'',i.note||'',i.fromTxnId||'',i.category||'']));
      writeSheet(ss, SAVINGS_SHEET, ['id','name','type','person','balance','baseBalance','goal','currency','balanceEquiv'],
        (data.savAccts||[]).map(a => [a.id,a.name,a.type,a.person||'',a.balance||0,a.baseBalance||0,a.goal||'',a.currency||'',a.balanceEquiv||'']));
      writeSheet(ss, CONTRIB_SHEET, ['id','date','amount','accountId','type','person','note','fromTxnId'],
        (data.contribs||[]).map(c => [c.id,c.date,c.amount,c.accountId,c.type,c.person||'',c.note||'',c.fromTxnId||'']));
      writeSheet(ss, RENTAL_SHEET, ['id','date','amount','cat','person','card','desc','note'],
        (data.rentalExpenses||[]).map(r => [r.id,r.date,r.amount,r.cat,r.person||'',r.card||'',r.desc||'',r.note||'']));
      writeSheet(ss, CATS_SHEET, ['key','label','color','fixed','budget','keywords'],
        (data.cats||[]).map(c => [c.key,c.label,c.color,c.fixed?'true':'false',c.budget||0,JSON.stringify(c.kw||[])]));
      try { const settings = data.settings || {}; writeSheet(ss, SETTINGS_SHEET, ['key','value'], Object.entries(settings).map(([k,v]) => [k, typeof v === 'string' ? v : JSON.stringify(v)])); } catch(e) {}
      return buildResponse({ok:true, count:(data.txns||[]).length});
    }

    if (data.action === 'pull') {
      const txns     = readSheet(ss, SHEET_NAME,   r => ({id:r.id,date:r.date,merchant:r.merchant,amount:parseFloat(r.amount)||0,category:r.category,person:r.person,card:r.card||'',note:r.note,refund:r.refund===true||r.refund==='true',manual:r.manual===true||r.manual==='true',incomeType:r.incomeType||undefined,savingsAccountId:r.savingsAccountId||undefined,incomeCategory:r.incomeCategory||undefined,needs:(r.needs===true||r.needs==='true')?true:(r.needs===false||r.needs==='false')?false:undefined,rental:(r.rental===true||r.rental==='true')?true:undefined,label:r.label||undefined,recurring:(r.recurring===true||r.recurring==='true')?true:undefined}));
      const learned  = {}; readSheet(ss, LEARNED_SHEET, r => r).forEach(r => { if(r.merchant) learned[r.merchant]=r.category; });
      const incomes  = readSheet(ss, INCOME_SHEET,  r => ({id:r.id,date:r.date,amount:parseFloat(r.amount)||0,type:r.type,person:r.person,desc:r.desc,note:r.note,fromTxnId:r.fromTxnId||undefined,category:r.category||undefined}));
      const savAccts = readSheet(ss, SAVINGS_SHEET, r => ({id:r.id,name:r.name,type:r.type,person:r.person,balance:parseFloat(r.balance)||0,baseBalance:parseFloat(r.baseBalance)||0,goal:r.goal?parseFloat(r.goal):null,currency:r.currency||'',balanceEquiv:r.balanceEquiv?parseFloat(r.balanceEquiv):null}));
      const contribs = readSheet(ss, CONTRIB_SHEET, r => ({id:r.id,date:r.date,amount:parseFloat(r.amount)||0,accountId:r.accountId,type:r.type,person:r.person,note:r.note,fromTxnId:r.fromTxnId||undefined}));
      const rentalExpenses = readSheet(ss, RENTAL_SHEET, r => ({id:r.id,date:r.date,amount:parseFloat(r.amount)||0,cat:r.cat,person:r.person,card:r.card||'',desc:r.desc,note:r.note}));
      const catsRaw = readSheet(ss, CATS_SHEET, r => { try { return {key:r.key,label:r.label,color:r.color,fixed:r.fixed==='true',budget:parseFloat(r.budget)||0,kw:JSON.parse(r.keywords||'[]')}; } catch(e) { return {key:r.key,label:r.label,color:r.color,fixed:r.fixed==='true',budget:parseFloat(r.budget)||0,kw:[]}; } });
      let settings = {};
      try { const settingsRaw = readSheet(ss, SETTINGS_SHEET, r => r); settingsRaw.forEach(r => { if (r.key) { try { settings[r.key] = JSON.parse(r.value); } catch(e) { settings[r.key] = r.value; } } }); } catch(e) {}
      return buildResponse({ok:true, txns, learned, incomes, savAccts, contribs, rentalExpenses, cats:catsRaw.length?catsRaw:null, settings});
    }

    return buildResponse({ok:false, error:'Unknown action'});
  } catch(err) {
    return buildResponse({ok:false, error:err.toString()});
  } finally {
    lock.releaseLock();
  }
}

function writeSheet(ss, name, headers, rows) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  sheet.clearContents();
  const data = rows.length ? [headers, ...rows] : [headers];
  sheet.getRange(1, 1, data.length, headers.length).setValues(data);
}

function readSheet(ss, name, mapper) {
  const sheet = ss.getSheetByName(name);
  if (!sheet) return [];
  const rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  const headers = rows[0];
  return rows.slice(1).map(r => {
    const obj = {};
    headers.forEach((h,i) => {
      const v = r[i];
      // Convert Google Sheets Date objects to YYYY-MM-DD strings
      if (v instanceof Date) {
        const y = v.getFullYear();
        const mo = String(v.getMonth()+1).padStart(2,'0');
        const d = String(v.getDate()).padStart(2,'0');
        obj[h] = y + '-' + mo + '-' + d;
      } else {
        obj[h] = v;
      }
    });
    return mapper(obj);
  }).filter(r => r.id || r.key);
}

function buildResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  return buildResponse({ok:true, status:'Ledger connected'});
}

Quick start

  1. Add your first transaction. Click the ＋ Add button on Transactions, or hit Upload Data to bring in a CSV or PDF statement.
  2. Review the preview. If you upload a statement, Ledger shows the parsed rows and a total so you can confirm it matches before importing.
  3. Categorise. Ledger auto-categorises using keyword rules. You can edit categories inline — it learns from your corrections.
  4. Sync (optional). Connect a Google Sheet to back up and sync across devices.

Features at a glance

  • Dashboard — monthly, 3-month, year-to-date and full-year views with category tiles, spending pace, and trend arrows.
  • Transactions — searchable, filterable, bulk-edit. Multi-select to re-categorise or delete in one go.
  • Income — log salary, transfers from friends, or any offset against a category.
  • Savings — track multiple accounts. Balances are always derived from your contribution log (never edited directly).
  • Mortgages — amortisation schedule, voluntary payments, balance history.
  • Budget targets — set a monthly target per category; tiles show a progress bar.
  • Recurring detection — Monthly Commitments card auto-surfaces subscriptions and regulars.
  • Duplicate checker — scans for suspiciously similar transactions before they clutter your history.
  • Dark mode — toggle in the topbar. Works everywhere.

Google Sheets sync

Ledger does not store your transactions. If you want data to survive browser clears or sync across devices, connect your own Google Sheet.

The full step-by-step (with the Apps Script code to paste) lives in Settings → Google Sheets Setup Guide.

Important — who can access your data: Even though the Apps Script deployment is set to Anyone (required for the app to call it), that only means the URL doesn't require Google login. The real access control is who has the URL and who you've shared the Google Sheet with. Only share the deployment URL and the sheet itself with people you actually want to have your data — your partner, for example. Treat the URL like a password.

FAQ

Where is my data stored?

Everything lives in your browser's localStorage. If you connect a Google Sheet, a copy also lives there (in your Drive).

What happens if I clear my browser or switch devices?

If you haven't connected Google Sheets, data is lost. With Sheets connected, pull from the sheet on the new device and everything is restored.

My CSV or PDF import looks wrong — what do I do?

Use the preview in the import modal to spot bad rows before committing. The total at the bottom should match your statement. If it doesn't, try remapping the columns (for CSV) or switching between bank/credit statement type (for PDF).

Can I edit a category after the fact?

Yes. Click the category badge on any transaction row to change it. Ledger learns from repeated corrections of the same merchant.

Why can't I edit a savings balance directly?

Balances are always derived from your contribution log. This prevents drift. Add a contribution (positive or negative) if you need to adjust.

How do I handle shared expenses with a partner?

Log the transaction once, then log an offsetting income entry under Income → Transfer with the partner as the person and the original category as the offset. The net effect: you only see your share.

Can I export my data?

Yes. Connect Google Sheets — the sheet that syncs is your permanent export by design. You can download it as CSV/Excel from Sheets any time.

Does Ledger work offline?

Yes for all local features. Google Sheets sync requires a connection.

Still stuck?

If something looks off, check Settings → Google Sheets Setup Guide or re-read the quick start above. Most issues are resolved by remapping CSV columns or re-deploying the Apps Script.
